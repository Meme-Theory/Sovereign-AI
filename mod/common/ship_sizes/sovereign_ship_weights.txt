# SOVEREIGN AI Overhaul — Ship Size AI Weights
# Controls fleet composition preferences for the AI.
#
# NOTE: This file adds ai_weight modifiers to existing ship sizes.
# Ship sizes themselves are defined in vanilla; we override the weight blocks.
#
# PHILOSOPHY:
#   - Early game: corvette swarms
#   - Mid game: balanced fleet with destroyers and cruisers
#   - Late game: battleship-heavy with corvette screen
#   - Adjust based on empire personality and current threat level

# ============================================================
# SCRIPTED VARIABLES
# ============================================================
@ship_base_weight = 100
@ship_early_game_years = 30
@ship_mid_game_years = 60
@ship_late_game_years = 100

# ============================================================
# CORVETTE
# ============================================================
# Fast, cheap, expendable. Core of early fleets, screening role later.

corvette = {
	max_speed = 5.0
	acceleration = 0.35
	rotation_speed = 0.1
	collision_radius = 2.0

	modifier = {
		ship_evasion_add = 60
	}

	max_hitpoints = 300
	size_multiplier = 1
	fleet_slot_size = 1
	section_slots = { "mid" = { locator = "part1" } }
	num_target_locators = 2
	is_space_station = no
	icon = ship_size_military_1
	base_buildtime = 60
	can_have_federation_design = yes
	enable_default_design = yes
	default_behavior = swarm

	combat_disengage_chance = 1.5

	class = shipclass_military
	construction_type = starbase_shipyard
	required_component_set = "power_core"
	required_component_set = "ftl_components"
	required_component_set = "thruster_components"
	required_component_set = "sensor_components"
	required_component_set = "combat_computers"

	resources = {
		category = ships
		cost = {
			alloys = 60
		}
		upkeep = {
			energy = 0.4
			alloys = 0.15
		}
	}

	ai_weight = {
		weight = @ship_base_weight

		# Early game: corvettes are king
		modifier = {
			factor = 2.0
			years_passed < @ship_early_game_years
		}

		# Late game: reduce corvette reliance
		modifier = {
			factor = 0.5
			years_passed > @ship_late_game_years
			owner = { has_technology = tech_battleships }
		}

		# Always maintain some screening corvettes
		modifier = {
			factor = 1.25
			owner = { has_technology = tech_battleships }
		}
	}
}
