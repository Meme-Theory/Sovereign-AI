# ============================================================
# EMPIRE TYPE CHECKS
# ============================================================

# Is a standard (non-gestalt, non-fallen) empire
sovereign_is_standard_empire = {
	is_regular_empire = yes
	is_fallen_empire = no
}

# Is a militaristic empire (any level)
sovereign_is_militaristic = {
	OR = {
		has_ethic = ethic_militarist
		has_ethic = ethic_fanatic_militarist
	}
}

# ============================================================
# ECONOMIC STATE CHECKS
# ============================================================

# Empire has a healthy economy
sovereign_has_stable_economy = {
	has_monthly_income = {
		resource = energy
		value > 10
	}
	has_monthly_income = {
		resource = minerals
		value > 10
	}
	has_resource = { type = energy amount > 200 }
	has_resource = { type = minerals amount > 200 }
}

# Empire is in economic crisis
sovereign_economy_is_critical = {
	OR = {
		has_monthly_income = {
			resource = energy
			value < -10
		}
		has_monthly_income = {
			resource = minerals
			value < -10
		}
		has_resource = { type = energy amount < 50 }
		has_resource = { type = minerals amount < 50 }
	}
}

# Empire has a strong alloy surplus for military production
sovereign_has_alloy_surplus = {
	has_monthly_income = {
		resource = alloys
		value > 20
	}
	has_resource = { type = alloys amount > 500 }
}

# Empire needs more consumer goods
sovereign_needs_consumer_goods = {
	is_regular_empire = yes
	has_monthly_income = {
		resource = consumer_goods
		value < 5
	}
}

# ============================================================
# MILITARY STATE CHECKS
# ============================================================

# Empire has a significant fleet
sovereign_has_strong_fleet = {
	used_naval_capacity_percent > 0.7
}

# Empire's fleet is depleted
sovereign_fleet_is_weak = {
	used_naval_capacity_percent < 0.3
}

# Empire is under significant threat
sovereign_is_threatened = {
	highest_threat > 50
}

# ============================================================
# PLANET CHECKS (planet scope)
# ============================================================

# Planet has room to grow
sovereign_planet_has_growth_room = {
	free_housing > 3
	free_district_slots > 0
}

# Planet is overcrowded
sovereign_planet_is_overcrowded = {
	free_housing < 0
	num_pops > 20
}

# Planet needs more jobs
sovereign_planet_needs_jobs = {
	num_unemployed > 3
	free_building_slots > 0
}
