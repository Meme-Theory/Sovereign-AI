# SOVEREIGN AI Overhaul — AI Personality Definitions
# Personalities define the strategic archetype of AI empires.
# Each personality has behavior flags, numeric modifiers, and spawn conditions.
#
# DESIGN PHILOSOPHY:
#   - Each personality should produce distinct, recognizable behavior
#   - Behavior flags must be coherent (no pacifist conquerors)
#   - Numeric modifiers should stay in reasonable ranges:
#     aggressiveness: 0.0–2.0
#     trade_willingness: 0.0–1.0
#     bravery: 0.5–1.5
#     military_spending: 0.5–2.0
#     colony_spending: 0.5–2.0

# ============================================================
# AGGRESSIVE EXPANSIONIST
# ============================================================
# Classic warmonger: conquers, subjugates, and dominates.
# Spawns primarily from militarist + xenophobe empires.

aggressive_expansionist = {
	weight_modifier = {
		weight = 10
		modifier = {
			factor = 2
			has_ethic = ethic_militarist
		}
		modifier = {
			factor = 2
			has_ethic = ethic_fanatic_militarist
		}
		modifier = {
			factor = 1.5
			has_ethic = ethic_xenophobe
		}
		modifier = {
			factor = 0
			has_ethic = ethic_pacifist
		}
		modifier = {
			factor = 0
			has_ethic = ethic_fanatic_pacifist
		}
	}

	allow = {
		is_country_type = default
		NOT = { has_ethic = ethic_pacifist }
		NOT = { has_ethic = ethic_fanatic_pacifist }
	}

	archetype = expansionist

	behaviour = {
		conqueror = yes
		subjugator = yes
		liberator = no
		opportunist = yes
		slaver = no
		uplifter = no
		purger = no
		dominator = yes
		infiltrator = yes
		robot_exploiter = yes
		robot_liberator = no
		migrator = no
	}

	aggressiveness = 1.5
	trade_willingness = 0.5
	bravery = 1.2
	military_spending = 1.5
	colony_spending = 1.2

	alliance_acceptance = -50
	federation_acceptance = -50

	threat_modifier = 1.25
	friction_modifier = 1.5
}
